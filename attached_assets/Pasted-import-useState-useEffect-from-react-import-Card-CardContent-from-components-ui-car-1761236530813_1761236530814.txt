export default function Testimonials() {
  const [index, setIndex] = useState(0);
  const [open, setOpen] = useState(false);
  const t = testimonials[index];

  const next = () => setIndex((i) => (i + 1) % testimonials.length);
  const prev = () => setIndex((i) => (i === 0 ? testimonials.length - 1 : i - 1));

  useEffect(() => {
    const handleKey = (e: KeyboardEvent) => {
      if (e.key === "ArrowRight") next();
      if (e.key === "ArrowLeft") prev();
    };
    window.addEventListener("keydown", handleKey);
    return () => window.removeEventListener("keydown", handleKey);
  }, []);

  return (
    <section className="py-16 bg-white">
      <div className="max-w-5xl mx-auto px-4 text-center">
        <h2 className="text-3xl md:text-4xl font-bold mb-8">Trusted by marketplace builders</h2>
        <Card className="rounded-2xl shadow-lg">
          <CardContent className="p-8 md:p-10">
            <p className="text-lg md:text-xl italic text-gray-900">“{t.quote}”</p>
            <div className="mt-6 flex flex-col items-center">
              <img src={t.image} alt={t.name} className="w-12 h-12 rounded-full mb-2 object-cover" />
              <p className="font-semibold">{t.name}</p>
              <p className="text-sm text-gray-500">
                {t.title}, {t.company} · {t.city}
              </p>
              <button
                onClick={() => setOpen(true)}
                className="text-blue-600 text-xs underline mt-1"
              >
                ✔ Verified customer — Read case study
              </button>
            </div>
            <div className="flex justify-center gap-3 mt-6">
              <Button variant="outline" size="sm" onClick={prev}>‹</Button>
              <Button variant="outline" size="sm" onClick={next}>›</Button>
            </div>
          </CardContent>
        </Card>

        <div className="flex justify-center mt-4 space-x-2">
          {testimonials.map((_, i) => (
            <span
              key={i}
              className={`h-2 w-2 rounded-full ${i === index ? "bg-blue-600" : "bg-gray-300"}`}
            ></span>
          ))}
        </div>
      </div>

      <Dialog open={open} onOpenChange={setOpen}>
        <DialogContent>
          <DialogHeader>
            <DialogTitle>{t.tagline}</DialogTitle>
            <DialogDescription>{t.blurb}</DialogDescription>
          </DialogHeader>
          <p className="text-sm text-gray-600 mt-2">{t.verifiedNote}</p>
        </DialogContent>
      </Dialog>
    </section>
  );
}
